    // Drum Synths

    function playDrum(type, time = 0) {

      if (!audioCtx) return;

      const t = time || audioCtx.currentTime;

      if (type === 'kick') {

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();

        osc.frequency.setValueAtTime(140, t);

        osc.frequency.exponentialRampToValueAtTime(0.01, t + 0.35);

        gain.gain.setValueAtTime(1, t);

        gain.gain.exponentialRampToValueAtTime(0.001, t + 0.35);

        osc.connect(gain);

        gain.connect(drumGain || masterGain);

        osc.start(t);

        osc.stop(t + 0.35);

      } else if (type === 'snare') {

        const bufferSize = audioCtx.sampleRate * 0.2;

        const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);

        const data = buffer.getChannelData(0);

        for (let i = 0; i < bufferSize; i++) data[i] = Math.random() * 2 - 1;

        const noise = audioCtx.createBufferSource();

        noise.buffer = buffer;

        const filter = audioCtx.createBiquadFilter();

        filter.type = 'highpass';

        filter.frequency.value = 1000;

        const noiseGain = audioCtx.createGain();

        noiseGain.gain.setValueAtTime(0.8, t);

        noiseGain.gain.exponentialRampToValueAtTime(0.01, t + 0.2);

        noise.connect(filter);

        filter.connect(noiseGain);

        noiseGain.connect(drumGain || masterGain);

        const osc = audioCtx.createOscillator();

        const oscGain = audioCtx.createGain();

        osc.type = 'triangle';

        osc.frequency.setValueAtTime(180, t);

        osc.frequency.exponentialRampToValueAtTime(40, t + 0.1);

        oscGain.gain.setValueAtTime(0.7, t);

        oscGain.gain.exponentialRampToValueAtTime(0.01, t + 0.15);

        osc.connect(oscGain);

        oscGain.connect(drumGain || masterGain);

        noise.start(t);

        osc.start(t);

        noise.stop(t + 0.2);

        osc.stop(t + 0.15);

      } else if (type === 'hihat') {

        const bufferSize = audioCtx.sampleRate * 0.05;

        const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);

        const data = buffer.getChannelData(0);

        for (let i = 0; i < bufferSize; i++) data[i] = Math.random() * 2 - 1;

        const noise = audioCtx.createBufferSource();

        noise.buffer = buffer;

        const filter = audioCtx.createBiquadFilter();

        filter.type = 'highpass';

        filter.frequency.value = 7000;

        const gain = audioCtx.createGain();

        gain.gain.setValueAtTime(0.5, t);

        gain.gain.exponentialRampToValueAtTime(0.001, t + 0.05);

        noise.connect(filter);

        filter.connect(gain);

        gain.connect(drumGain || masterGain);

        noise.start(t);

        noise.stop(t + 0.05);

      } else if (type === 'clap') {

        const bufferSize = audioCtx.sampleRate * 0.25;

        const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);

        const data = buffer.getChannelData(0);

        for (let i = 0; i < bufferSize; i++) data[i] = Math.random() * 2 - 1;

        const noise = audioCtx.createBufferSource();

        noise.buffer = buffer;

        const filter = audioCtx.createBiquadFilter();

        filter.type = 'bandpass';

        filter.frequency.value = 1200;

        const gain = audioCtx.createGain();

        gain.gain.setValueAtTime(0.7, t);

        gain.gain.exponentialRampToValueAtTime(0.01, t + 0.25);

        noise.connect(filter);

        filter.connect(gain);

        gain.connect(drumGain || masterGain);

        noise.start(t);

        noise.stop(t + 0.25);

      }

    }

    // Metronome Click

    function playClick(time, isDownbeat) {

      if (!isMetronomeOn || !audioCtx) return;

      const osc = audioCtx.createOscillator();

      const gain = audioCtx.createGain();

      osc.type = 'sine';

      osc.frequency.value = isDownbeat ? 1200 : 800;

      gain.gain.setValueAtTime(0.4, time);

      gain.gain.exponentialRampToValueAtTime(0.0001, time + 0.04);

      osc.connect(gain);

      gain.connect(metroGain);

      osc.start(time);

      osc.stop(time + 0.05);

    }

    // Synthesize warm vocal reverb impulse

    function createReverbImpulse(duration = 2.0, decay = 2.5) {

      const sampleRate = audioCtx.sampleRate;

      const length = sampleRate * duration;

      const impulse = audioCtx.createBuffer(2, length, sampleRate);

      const left = impulse.getChannelData(0);

      const right = impulse.getChannelData(1);

      for (let i = 0; i < length; i++) {

        const n = i;

        left[i] = (Math.random() * 2 - 1) * Math.pow(1 - n / length, decay);

        right[i] = (Math.random() * 2 - 1) * Math.pow(1 - n / length, decay);

      }

      return impulse;

    }

    // Initialize Audio Engine & Vocal Processing Chain

    async function initAudio() {

      if (audioCtx) {

        if (audioCtx.state === 'suspended') await audioCtx.resume();

        document.getElementById('startModal').classList.add('hidden');

        return;

      }

      const AudioContextClass = window.AudioContext || window.webkitAudioContext;

      audioCtx = new AudioContextClass();

      if (audioCtx.state === 'suspended') await audioCtx.resume();

     

      masterGain = audioCtx.createGain();

      masterGain.gain.value = 1.0;

      masterGain.connect(audioCtx.destination);

      metroGain = audioCtx.createGain();

      metroGain.gain.value = 0.8;

      metroGain.connect(masterGain);

      drumGain = audioCtx.createGain();

      drumGain.gain.value = 0.9;

      drumGain.connect(masterGain);

      // Setup Vocal FX Routing

      vocalPreGain = audioCtx.createGain();

      vocalPreGain.gain.value = 1.0;

      vocalHpFilter = audioCtx.createBiquadFilter();

      vocalHpFilter.type = 'highpass';

      vocalHpFilter.frequency.value = 110; // Filter low mic rumbles

      // Reverb FX

      vocalReverbConvolver = audioCtx.createConvolver();

      vocalReverbConvolver.buffer = createReverbImpulse(2.2, 2.8);

      vocalReverbGain = audioCtx.createGain();

      vocalReverbGain.gain.value = 0.3;

      // Delay / Echo FX

      vocalDelayNode = audioCtx.createDelay();

      vocalDelayNode.delayTime.value = 0.32; // ~320ms echo

      vocalDelayFeedback = audioCtx.createGain();

      vocalDelayFeedback.gain.value = 0.35;

      vocalDelayGain = audioCtx.createGain();

      vocalDelayGain.gain.value = 0.2;

      vocalDelayNode.connect(vocalDelayFeedback);

      vocalDelayFeedback.connect(vocalDelayNode);

      vocalDelayNode.connect(vocalDelayGain);

      // In-Ear Monitoring Gain (Default 0 / Off to prevent feedback)

      vocalMonitorGain = audioCtx.createGain();

      vocalMonitorGain.gain.value = 0;

      vocalMonitorGain.connect(masterGain);

      // Mic Stream setup

      try {

        micStream = await navigator.mediaDevices.getUserMedia({

          audio: {

            echoCancellation: false,

            noiseSuppression: false,

            autoGainControl: false

          }

        });

      } catch (err) {

        console.warn("Falling back to default mic constraints:", err);

        try {

          micStream = await navigator.mediaDevices.getUserMedia({ audio: true });

        } catch (e) {

          alert("לא התקבלה הרשאת מיקרופון בדפדפן. לא ניתן להקליט ללא מיקרופון.");

        }

      }

      if (micStream) {

        micSource = audioCtx.createMediaStreamSource(micStream);

        window._micStreamRef = micStream;

        window._micSourceRef = micSource;

        // Wire mic into Vocal Chain

        micSource.connect(vocalPreGain);

        vocalPreGain.connect(vocalHpFilter);

        // Vocal FX Branch

        vocalHpFilter.connect(vocalReverbConvolver);

        vocalReverbConvolver.connect(vocalReverbGain);

        vocalReverbGain.connect(vocalMonitorGain);

        vocalHpFilter.connect(vocalDelayNode);

        vocalDelayGain.connect(vocalMonitorGain);

        // Direct dry path to monitor

        vocalHpFilter.connect(vocalMonitorGain);

      }

      initLooperRecorder();

      // Start Scheduler Clock

      nextBeatTime = audioCtx.currentTime + 0.05;

      scheduler();

      document.getElementById('startModal').classList.add('hidden');

    }

    function initLooperRecorder() {

      recorderNode = audioCtx.createScriptProcessor(4096, 1, 1);

      window._recorderNodeRef = recorderNode;

     

      recorderNode.onaudioprocess = function(e) {

        if (!micSource) return;

        const input = e.inputBuffer.getChannelData(0);

       

        // Push raw samples to active recording loop tracks

        tracks.forEach(track => {

          if (track.state === 'recording') {

            const chunk = new Float32Array(input.length);

            chunk.set(input);

            track.recordedChunks.push(chunk);

            track.recordedSamples += input.length;

          }

        });

        // Push samples to Master Song Recording if active

        if (isSongRecording) {

          const chunk = new Float32Array(input.length);

          chunk.set(input);

          songRecChunks.push(chunk);

          songRecSamples += input.length;

        }

      };

      if (micSource) {

        micSource.connect(recorderNode);

        const silentGain = audioCtx.createGain();

        silentGain.gain.value = 0;

        recorderNode.connect(silentGain);

        silentGain.connect(audioCtx.destination);

      }

    }

    // Initialize Tracks Array

    for (let i = 0; i < NUM_TRACKS; i++) {

      tracks.push({

        id: i,

        name: `ערוץ ${i + 1}`,

        state: 'empty', // 'empty', 'armed', 'recording', 'playing'

        isMuted: false,

        isSolo: false,

        volume: 0.9,

        gainNode: null,

        audioBuffer: null,

        currentSource: null,

        recordedChunks: [],

        recordedSamples: 0

      });

    }

    // High Precision Web Audio Scheduler

    const lookahead = 25.0; // ms

    const scheduleAheadTime = 0.1; // seconds

    function scheduler() {

      if (!isMasterPlaying || !audioCtx) {

        timerId = setTimeout(scheduler, lookahead);

        return;

      }

      while (nextBeatTime < audioCtx.currentTime + scheduleAheadTime) {

        scheduleBeat(currentBeat, nextBeatTime);

        advanceBeat();

      }

      timerId = setTimeout(scheduler, lookahead);

    }

    function advanceBeat() {

      const secondsPerBeat = 60.0 / bpm;

      nextBeatTime += secondsPerBeat;

      currentBeat = (currentBeat + 1) % totalBeats;

    }

    function scheduleBeat(beatNumber, time) {

      const isCycleStart = beatNumber === 0;

      const beatInBar = beatNumber % beatsPerBar;

      // Click

      playClick(time, beatInBar === 0);

      // Auto Drum Beat

      if (isAutoDrumOn) {

        playAutoDrumPattern(beatNumber, time);

      }

      // Handle Quantized Track Recording & Playback on Cycle Start

      if (isCycleStart) {

        const loopDuration = (60.0 / bpm) * totalBeats;

       

        tracks.forEach(track => {

          if (track.state === 'recording') {

            finalizeRecording(track, loopDuration);

          }

          if (track.state === 'armed') {

            track.state = 'recording';

            track.recordedChunks = [];

            track.recordedSamples = 0;

            updateTrackUI(track.id);

          }

          if (track.state === 'playing' && track.audioBuffer) {

            playTrackLoop(track, time, loopDuration);

          }

        });

      }

      // Visual UI update

      const delay = Math.max(0, (time - audioCtx.currentTime) * 1000);

      setTimeout(() => {

        highlightBeatUI(beatNumber);

      }, delay);

    }

    function playAutoDrumPattern(beatIndex, time) {

      const beatInBar = beatIndex % beatsPerBar;

      const secondsPerBeat = 60.0 / bpm;

     

      // Kick on beat 1 and 3

      if (beatInBar === 0 || beatInBar === 2) {

        playDrum('kick', time);

      }

      // Snare / Clap on beat 2 and 4

      if (beatInBar === 1 || beatInBar === 3) {

        playDrum('snare', time);

      }

      // Hi-Hat on every 8th note

      playDrum('hihat', time);

      playDrum('hihat', time + (secondsPerBeat / 2));

    }

    function playTrackLoop(track, time, loopDuration) {

      if (!track.audioBuffer || !audioCtx) return;

     

      if (!track.gainNode) {

        track.gainNode = audioCtx.createGain();

        track.gainNode.connect(masterGain);

      }

     

      applyTrackVolume(track);

      const source = audioCtx.createBufferSource();

      source.buffer = track.audioBuffer;

      source.connect(track.gainNode);

      source.start(time);

      track.currentSource = source;

    }

    function finalizeRecording(track, loopDuration) {

      const totalRecorded = track.recordedSamples;

      if (totalRecorded === 0) {

        track.state = 'empty';

        updateTrackUI(track.id);

        return;

      }

      const flat = new Float32Array(totalRecorded);

      let offset = 0;

      for (let chunk of track.recordedChunks) {

        flat.set(chunk, offset);

        offset += chunk.length;

      }

      const targetLength = Math.floor(audioCtx.sampleRate * loopDuration);

      const audioBuffer = audioCtx.createBuffer(1, targetLength, audioCtx.sampleRate);

      const channelData = audioBuffer.getChannelData(0);

      for (let i = 0; i < targetLength; i++) {

        channelData[i] = i < flat.length ? flat[i] : 0;

      }

      track.audioBuffer = audioBuffer;

      track.state = 'playing';

      track.recordedChunks = [];

      updateTrackUI(track.id);

    }

    function applyTrackVolume(track) {

      if (!track.gainNode) return;

      const anySolo = tracks.some(t => t.isSolo);

      if (track.isMuted || (anySolo && !track.isSolo)) {

        track.gainNode.gain.value = 0;

      } else {

        track.gainNode.gain.value = track.volume;

      }

    }

    async function processSongMixdown() {

      if (songRecSamples === 0) {

        document.getElementById('songRecTimer').textContent = 'לא נקלט אודיו בהקלטה.';

        return;

      }

      // Flatten vocal input

      const vocalFlat = new Float32Array(songRecSamples);

      let offset = 0;

      for (let chunk of songRecChunks) {

        vocalFlat.set(chunk, offset);

        offset += chunk.length;

      }

      const durationSeconds = songRecSamples / audioCtx.sampleRate;

      const offlineCtx = new OfflineAudioContext(2, Math.floor(44100 * durationSeconds), 44100);

      // Render all active loops continuously over this duration

      const loopDuration = (60.0 / bpm) * totalBeats;

      const activeTracks = tracks.filter(t => t.audioBuffer && !t.isMuted);

      activeTracks.forEach(track => {

        let startTime = 0;

        while (startTime < durationSeconds) {

          const src = offlineCtx.createBufferSource();

          src.buffer = track.audioBuffer;

          const g = offlineCtx.createGain();

          g.gain.value = track.volume;

          src.connect(g);

          g.connect(offlineCtx.destination);

          src.start(startTime);

          startTime += loopDuration;

        }

      });

      // Render Vocal track onto offline context

      const vocalBuffer = offlineCtx.createBuffer(1, songRecSamples, audioCtx.sampleRate);

      vocalBuffer.getChannelData(0).set(vocalFlat);

      const vocalSrc = offlineCtx.createBufferSource();

      vocalSrc.buffer = vocalBuffer;

      const vocalGain = offlineCtx.createGain();

      vocalGain.gain.value = 1.0;

      vocalSrc.connect(vocalGain);

      vocalGain.connect(offlineCtx.destination);

      vocalSrc.start(0);

      const renderedSong = await offlineCtx.startRendering();

      lastRenderedSongBlob = audioBufferToWav(renderedSong);

      // Update UI Player

      const url = URL.createObjectURL(lastRenderedSongBlob);

      const player = document.getElementById('vocalTakeAudioPlayer');

      player.src = url;

      document.getElementById('vocalTakeContainer').classList.remove('hidden');

      document.getElementById('songRecTimer').textContent = 'טייק השיר מוכן להאזנה ולהורדה!';

    }

    function tapTempo() {

      const now = performance.now();

      tapTimes.push(now);

      if (tapTimes.length > 4) tapTimes.shift();

      if (tapTimes.length >= 2) {

        let diffs = [];

        for (let i = 1; i < tapTimes.length; i++) {

          diffs.push(tapTimes[i] - tapTimes[i-1]);

        }

        const avgDiff = diffs.reduce((a, b) => a + b, 0) / diffs.length;

        const calculatedBpm = Math.round(60000 / avgDiff);

        if (calculatedBpm >= 50 && calculatedBpm <= 220) {

          bpm = calculatedBpm;

          document.getElementById('bpmSlider').value = bpm;

          document.getElementById('bpmVal').textContent = bpm;

        }

      }

    }

    function audioBufferToWav(buffer) {

      const numChannels = buffer.numberOfChannels;

      const sampleRate = buffer.sampleRate;

      const format = 1; // PCM

      const bitDepth = 16;

     

      const bytesPerSample = bitDepth / 8;

      const blockAlign = numChannels * bytesPerSample;

     

      const length = buffer.length * blockAlign;

      const bufferArr = new ArrayBuffer(44 + length);

      const view = new DataView(bufferArr);

      function writeString(view, offset, string) {

        for (let i = 0; i < string.length; i++) {

          view.setUint8(offset + i, string.charCodeAt(i));

        }

      }

      writeString(view, 0, 'RIFF');

      view.setUint32(4, 36 + length, true);

      writeString(view, 8, 'WAVE');

      writeString(view, 12, 'fmt ');

      view.setUint32(16, 16, true);

      view.setUint16(20, format, true);

      view.setUint16(22, numChannels, true);

      view.setUint32(24, sampleRate, true);

      view.setUint32(28, sampleRate * blockAlign, true);

      view.setUint16(32, blockAlign, true);

      view.setUint16(34, bitDepth, true);

      writeString(view, 36, 'data');

      view.setUint32(40, length, true);

      let offset = 44;

      for (let i = 0; i < buffer.length; i++) {

        for (let ch = 0; ch < numChannels; ch++) {

          let sample = buffer.getChannelData(ch)[i];

          sample = Math.max(-1, Math.min(1, sample));

          sample = sample < 0 ? sample * 0x8000 : sample * 0x7FFF;

          view.setInt16(offset, sample, true);

          offset += 2;

        }

      }

      return new Blob([bufferArr], { type: 'audio/wav' });

    }
